server {
    server_name www.fazer.city fazer.city;

    root /opt/mro_dash/site;
    index index.html;

    # Proxy dynamic routes to Flask
    location /login {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /callback {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /api {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /me {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Static files
    location = /fcoin {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Pages légales servies directement depuis les fichiers HTML
    location = /mentions-legales { try_files /legal.html =404; }
    location = /cgu             { try_files /terms.html =404; }
    location = /terms           { try_files /terms.html =404; }
    location = /confidentialite { try_files /privacy.html =404; }
    location = /privacy         { try_files /privacy.html =404; }

    location /assets/ { try_files $uri =404; }
    location / { try_files $uri $uri/ /index.html; }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/fazer.city/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/fazer.city/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
